from probe import *

import os
BASE=next(r['fen'] for r in json.loads((ROOT/'screen.json').read_text()) if r['id']=='R2-093')

def run(task):
    line,name,mask,nodes,forced=task;b=chess.Board(BASE)
    for mv in line:b.push_uci(mv)
    b=chess.Board(b.fen())
    binary={'production':RIVAL,'hce':RIVAL,'prior_full_qdraw':'/home/chris/benchmark/sprt/rival-net1291A','prior_tt_depth':'/home/chris/benchmark/sprt/rival-net1291C'}.get(name,DIAG)
    with chess.engine.SimpleEngine.popen_uci(binary,env=dict(os.environ,RIVAL_DIAG_DISABLE=str(mask))) as e:
        e.configure({'Threads':1,'Hash':128})
        if name=='hce':e.configure({'UseNNUE':False})
        kw={'root_moves':[chess.Move.from_uci(forced)]} if forced else {}
        r=encode(e.play(b,chess.engine.Limit(nodes=nodes),game=object(),info=chess.engine.INFO_ALL,**kw),b)
    return {'line':line,'fen':b.fen(),'name':name,'disabled_mask':mask,'budget_nodes':nodes,'forced':forced,**r}
