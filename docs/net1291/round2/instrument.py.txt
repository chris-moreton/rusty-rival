from pathlib import Path
import subprocess
root=Path('/home/chris/benchmark/net1291-worktree')
p=root/'src/search.rs'; s=p.read_text()
assert subprocess.check_output(['git','diff','--','src/search.rs','src/quiesce.rs'],cwd=root)==b''
helper='''
/// NET-1291 diagnosis only: independently disable a search mechanism.
/// No behaviour change or environment lookup in production builds.
#[inline(always)]
pub(crate) fn diagnostic_disabled(bit: u32) -> bool {
    #[cfg(feature = "search-width-diagnostics")]
    {
        static MASK: std::sync::LazyLock<u32> = std::sync::LazyLock::new(|| {
            std::env::var("RIVAL_DIAG_DISABLE").ok().and_then(|s| s.parse().ok()).unwrap_or(0)
        });
        *MASK & bit != 0
    }
    #[cfg(not(feature = "search-width-diagnostics"))]
    {
        let _ = bit;
        false
    }
}

'''
s=s.replace('/// Make a move with lazy NNUE tracking.',helper+'/// Make a move with lazy NNUE tracking.',1)
repls=[
